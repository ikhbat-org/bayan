  /// CoverageTests.cs:26 `Run`: `exe args` in `dir` with `BAYAN_COVERAGE` set to `prefix` (unset when null), through

  /// The hits of the true and the false counter of the `nth` branch on `snippet`'s line, as `true false`.

  // CoverageTests.cs:240 `ProbesCountWhatRan` (R9): statements, both outcomes of if, while, ternary, `&&` and `||`, and error
  // paths by (function, variant) with their origin.
  test "coverage t30h: ProbesCountWhatRan" {
      if Target.os == "windows" {
          todo("the pid is read through `sh -c 'echo $$; exec'`, and Windows has no `sh`")
      }
      let dir = t30h4Dir("paths")
      let exe = t30h4Build(dir, by: t30h4Paths())
      let map = t30h4Map(Path.combine(dir, "main.byn"), by: false)
      let prefix = Path.combine(dir, "paths")
      var pid = ""
      let outcome = t30h4Run(exe, by: prefix, at: dir, for: List<string>(), through: ref pid)
      assert(outcome.stdout == "pos\nneg\n3\nfalse false\n1\n")
      let counts = t30h4Counts(prefix, for: pid)
      // Statements: one per statement, the main program's too.
      assert(t30h4Hits(counts, for: t30h4At(map, by: "line", for: "i += 1")) == 3)
      assert(t30h4Hits(counts, for: t30h4At(map, by: "line", for: "return \"pos\"")) == 1)
      assert(t30h4Hits(counts, for: t30h4At(map, by: "line", for: "print(sign(-1))")) == 1)
      assert(t30h4Hits(counts, for: t30h4At(map, by: "line", for: "return n\n")) == 0)
      // Branches: each outcome its own counter.
      assert(t30h4Branch(map, by: counts, for: "if n > 0") == "1 1")
      assert(t30h4Branch(map, by: counts, for: "while i < n") == "3 1")
      assert(t30h4Branch(map, by: counts, for: "n == 0 ?") == "0 1")
      assert(t30h4Branch(map, by: counts, for: "a && b", nth: 0) == "1 0")
      assert(t30h4Branch(map, by: counts, for: "a && b", nth: 1) == "0 1")
      assert(t30h4Branch(map, by: counts, for: "a || b", nth: 0) == "0 1")
      assert(t30h4Branch(map, by: counts, for: "a || b", nth: 1) == "0 1")
      assert(t30h4Branch(map, by: counts, for: "fail .bad if n < 0") == "1 0")
      // Error paths: `check` raises E.bad with `fail`, `pass` passes it on through `try`; E.worse never happens.
      var raised = List<emitter.CoverageCounter>()
      var passed = List<emitter.CoverageCounter>()
      for c in map.counters {
          continue if c.kind != "error"
          if c.function == "check" {
              raised.append(c)
          } else if c.function == "pass" {
              passed.append(c)
          }
      }
      assert(raised.length == 1)
      assert(raised[0].variant == "E.bad"
          && raised[0].origin == "fail"
          && raised[0].line == t30h4LineOf(t30h4Paths(), by: "fail .bad"))
      assert(t30h4Hits(counts, for: raised[0]) == 1)
      assert(passed.length == 2 && passed[0].variant == "E.bad" && passed[1].variant == "E.worse")
      for c in passed {
          assert(c.origin == "try" && c.line == t30h4LineOf(t30h4Paths(), by: "try check"))
      }
      assert(t30h4Hits(counts, for: passed[0]) == 1 && t30h4Hits(counts, for: passed[1]) == 0)
      deleteWorkDir(dir)
  }
  
  // CoverageTests.cs:283-286 `CountersFlushOnExitAndPanic` (R10): counters reach the file when the program stops through `exit()`
  // (`todo` exits 4) and on a panic (`byn_cov_flush`); one row per C# InlineData.
  @cases(("todo(\"later\")", 4), ("panic(\"boom\")", 134))
  test "coverage t30h: CountersFlushOnExitAndPanic" (string stop, i32 exitCode) {
      if Target.os == "windows" {
          todo("the pid is read through `sh -c 'echo $$; exec'`, and Windows has no `sh`")
      }
      let dir = t30h4Dir(exitCode == 4 ? "stop-todo" : "stop-panic")
      let exe = t30h4Build(dir, by: $"void step() {{\n    print(\"step\")\n}}\n\nstep()\nstep()\n{stop}\n")
      let map = t30h4Map(Path.combine(dir, "main.byn"), by: false)
      let prefix = Path.combine(dir, "stop")

          for c in map.counters {

                  at.append(c)
              }
          }
          assert(at.length == 1)

  test "coverage t30h: CounterFilesAreNamedByTestIndexOrPid" {
      if Target.os == "windows" {
          todo("the pid is read through `sh -c 'echo $$; exec'`, and Windows has no `sh`")
      }

      goldenWrite(root, for: "package.byn", as: "Package { name: \"local/tests\", version: \"0.1.0\" }\n")
      goldenWrite(
          root,
          for: "src/app/main.byn",
          as: "i64 twice(i64 n) {\n    return n * 2\n}\n\ntest \"one\" {\n    assert(twice(1) == 2)\n}\n\ntest \"two\" {\n    assert(twice(2) == 4)\n    assert(twice(3) == 6)\n}\n",
