  // The single progress line on stderr (Progress.cs, spec 005 R13). The C# draws it from a `Timer` once the operation outlives
  // 300 ms and every 250 ms after. The language has no timer thread, so `start` reads `Io.monotonicNanos()` and `done` draws the frames
  // the timer would have drawn by then, before it clears the line (Q1116).
  
  /// The line's state: the C# `Progress` fields `_frames`, `_phase`, `_frame`, `_shown`, `_done`; `off` stands for its `TextWriter.Null` writer.
  struct Progress {
      bool   off   = false
      bool   ascii = false
      string phase = ""
      i64    frame = 0
      bool   shown = false
      bool   done  = false
      /// `timed` is false for an inert `Progress {}` and when `after` is zero or less (the C# creates no `Timer`); `begun` is `Io.monotonicNanos()` at `start`, `after` the C# `threshold` in nanoseconds.
      bool   timed = false
      i64    begun = 0
      i64    after = 0
  }
  
  /// `Progress.DefaultThreshold` (Progress.cs:17) and `s_interval` (:18), in nanoseconds.
  const progressThreshold: i64 = 300000000
  const progressInterval: i64  = 250000000
  
  /// The scalar's width in UTF-8 bytes.
  i64 t7Utf8Length(i64 scalar) {
~     return 1 if scalar < 0x80
~     return 2 if scalar < 0x800
~     return 3 if scalar < 0x10000
~     return 4
  }
  
  /// TerminalStyle.cs:46 `Hidden` for one UTF-16 unit; a scalar above U+FFFF is never hidden.
  bool t7Hidden(i64 c) {
~     return c is < 0x20 or 0x7F..=0x9F or 0x61C or 0x200E or 0x200F or 0x202A..=0x202E or 0x2066..=0x2069
  }
  
  /// TerminalStyle.cs:34 `Visible`: C0, DEL, C1 and bidi controls written as `\u{hex}`.
  string visible(string text) uses alloc {
~     var any = false
~     for r in text.runes {
~         if t7Hidden(i64(u32(r))) {
~             any = true
          }
      }
~     unless any {
~         return text
      }
~     var sb = StrBuf()
~     var from: i64 = 0
~     var at: i64 = 0
~     for r in text.runes {
~         let scalar = i64(u32(r))
~         let next = at + t7Utf8Length(scalar)
~         if t7Hidden(scalar) {
~             sb.append(text[from..at])
~             sb.append("\\u{")
~             sb.append(string(scalar, as: .hex))
~             sb.append("}")
~             from = next
          }
~         at = next
      }
~     sb.append(text[from..text.length])
~     return $"{sb.finish()}"
  }
  
  /// Progress.cs:67 `Columns`: two per wide scalar, none per combining mark, one otherwise.
  i64 columns(i64 scalar) {
~     return 0 if scalar is 0x0300..=0x036F
~     return 2 if scalar is 0x1100..=0x115F
          or 0x2E80..=0xA4CF
          or 0xAC00..=0xD7A3
          or 0xF900..=0xFAFF
          or 0xFE30..=0xFE4F
          or 0xFF00..=0xFF60
          or 0xFFE0..=0xFFE6
          or 0x1F300..=0x1F64F
          or 0x1F900..=0x1F9FF
          or 0x20000..=0x3FFFD
~     return 1
  }
  
  /// Progress.cs:48 `Fit`: `phase` cut to `room` columns ending in `ellipsis`. The C# `ellipsis.Length` counts UTF-16 units: `length` is that count.
  string fit(string phase, to i64 room, by string ellipsis, for i64 length) uses alloc {
~     var width: i64 = 0
~     for r in phase.runes {
~         width += columns(i64(u32(r)))
      }
~     return phase if width <= room
~     let keep = room - length > 0 ? room - length : 0
~     var kept: i64 = 0
~     var used: i64 = 0
~     for r in phase.runes {
~         used += columns(i64(u32(r)))
~         break if used > keep
~         kept += 1
      }
~     return $"{phase.runes.take(kept)}{ellipsis}"
  }
  
  /// Progress.cs:23 `Progress` and :31 `Start`: the line, or an inert one when it must not be shown. `after` is the C# `threshold` in nanoseconds; zero or less draws at once.
  Progress start(in DStyle style, by bool isTty, through bool json, for string phase, i64 after = progressThreshold)
          uses alloc, clock, console, env {
~     let off = !isTty || json || (Env.get("CI") ?? "").length > 0
      // symbols.Progress[0].Length and symbols.Ellipsis.Length are 3 for ASCII ("...") and 3 and 1 for Unicode
~     let ellipsis = style.ascii ? "..." : "…"
~     let ellipsisLength: i64 = style.ascii ? 3 : 1
~     var p = Progress {
          off: off,
          ascii: style.ascii,
          phase: fit(visible(phase), to: style.width - 1 - 3 - 1, by: ellipsis, for: ellipsisLength),
          timed: after > 0,
          begun: Io.monotonicNanos(),
          after: after,
      }
~     if after <= 0 {
~         tick(ref p)
      }
~     return p
  }
  
  /// Progress.cs:77 `Tick`: draws the next frame; a single-frame (ASCII) line is drawn once.
  void tick(ref Progress p) uses alloc, console {
~     return if p.done
~     return if p.shown && p.ascii
~     let frames: List<string> = p.ascii ? ["..."] : ["◆◇◇", "◇◆◇", "◇◇◆"]
~     let text = $"\r{frames[p.frame % frames.length]} {p.phase}"
~     p.frame += 1
~     if !p.off {
~         Io.stderr.write(text)
      }
~     p.shown = true
  }
  
  /// Progress.cs:90 `Done`: clears the line; safe to call twice.
  void done(ref Progress p) uses alloc, clock, console {
~     return if p.done
~     if p.timed && !p.ascii {
          // the timer's firings so far: one at `after`, then one per interval; a single-frame (ASCII) line is drawn once, below
~         let elapsed = Io.monotonicNanos() - p.begun
~         if elapsed >= p.after {
~             let owed = 1 + (elapsed - p.after) / progressInterval - p.frame
~             var i: i64 = 0
~             while i < owed {
~                 tick(ref p)
~                 i += 1
              }
          }
~     } else if p.timed && Io.monotonicNanos() - p.begun >= p.after {
~         tick(ref p)
      }
~     p.done = true
~     return if !p.shown
~     if !p.off {
~         Io.stderr.write(p.ascii ? "\n" : "\r\u{1b}[2K")
      }
  }
  
  // ProgressTests: `isTty: false` keeps the line off stderr; `phase`, `frame` and `shown` are what the C# writer received.
  // ProgressTests.TheLineIsCutToTheTerminalWidth.
  test "progress: the line is cut to the terminal width" {
      let phase = $"testing {"w".repeated(60)}"
      var p = start(DStyle { width: 40 }, by: false, through: false, for: phase, after: 0)
      done(ref p)
      assert(p.phase == $"testing {"w".repeated(26)}…" && p.frame == 1 && p.shown && !p.ascii)
      var a = start(DStyle { ascii: true, width: 40 }, by: false, through: false, for: phase, after: 0)
      done(ref a)
      assert(a.phase == $"testing {"w".repeated(24)}..." && a.frame == 1 && a.shown && a.ascii)
      var s = start(DStyle { width: 40 }, by: false, through: false, for: "testing www", after: 0)
      done(ref s)
      assert(s.phase == "testing www" && s.frame == 1 && s.shown)
  }
  
  // ProgressTests.WideScalarsCountTwoColumnsAndMarksNone (width 40: room 40 - 1 - 3 - 1 = 35, keep 34 before the ellipsis).
  test "progress: wide scalars count two columns and marks none" {
      let style = DStyle { width: 40 }
      let cut = $"{"漢".repeated(17)}…"
      assert(start(style, by: false, through: false, for: "漢".repeated(30), after: 0).phase == cut)
      assert(start(style, by: false, through: false, for: $"{"漢".repeated(17)}xy", after: 0).phase == cut)
      let marked = $"{"e".repeated(35)}\u{301}"
      assert(start(style, by: false, through: false, for: marked, after: 0).phase == marked)
  }
  
  // ProgressTests.ControlCharactersInThePhaseAreShownEscaped.
  test "progress: control characters in the phase are shown escaped" {
      let p = start(DStyle {}, by: false, through: false, for: "testing y\u{1b}]0;PWNED\u{7}\u{1b}[2J", after: 0)
      assert(p.phase == "testing y\\u{1b}]0;PWNED\\u{7}\\u{1b}[2J")
  }
  
  // ProgressTests.UnicodeRewritesInPlaceAndClears.
  test "progress: unicode rewrites in place and clears" {
      var p = start(DStyle {}, by: false, through: false, for: "compiling", after: 0)
      assert(p.phase == "compiling" && p.frame == 1 && !p.ascii)
      done(ref p)
      assert(p.done && p.shown)
  }
  
  // ProgressTests.AsciiPrintsOnceWithoutRewriting.
  test "progress: ascii prints once without rewriting" {
      var p = start(DStyle { ascii: true }, by: false, through: false, for: "compiling", after: 0)
      tick(ref p)
      done(ref p)
      assert(p.phase == "compiling" && p.frame == 1 && p.ascii && p.shown)
  }
  
  // ProgressTests.NonTerminalPrintsNothing.
  test "progress: a non-terminal prints nothing" {
      assert(start(DStyle {}, by: false, through: false, for: "compiling", after: 0).off)
  }
  
  // ProgressTests.JsonModePrintsNothing.
  test "progress: json mode prints nothing" {
      assert(start(DStyle {}, by: true, through: true, for: "compiling", after: progressThreshold).off)
  }
  
  // ProgressTests.BelowThresholdPrintsNothing.
  test "progress: below the threshold prints nothing" {
      var p = start(DStyle {}, by: false, through: false, for: "compiling", after: 3600000000000)
      done(ref p)
      assert(p.done && !p.shown && p.frame == 0)
  }
  
  // ProgressTests.TimerCyclesFramesThenClears.
  test "progress: the timer cycles frames then clears" {
      var p = start(DStyle {}, by: false, through: false, for: "compiling", after: 1000000)
      while Io.monotonicNanos() - p.begun < 1000000 {}
      done(ref p)
      assert(p.frame >= 1 && p.shown && p.done && !p.ascii)
  }
