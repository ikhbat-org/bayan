  /// mutants 2, 5 and 7 exit 1 and mutant 4 sleeps past its limit (5 s after a mutant 0 of a few milliseconds); site 6 is marked and
  /// site 8 not emitted. Returns the directory.
  string t13JobsFixture(string name) uses alloc, fs {
~     let dir = Path.combine(Path.fullPath("tmp"), name)
~     deleteWorkDir(dir)
~     goldenWrite(dir, for: "prog", as: "")
~     var sites = StrBuf()
~     sites.append("{ \"version\": 1, \"sites\": [")
~     for id in 1..10 {
~         let mark = id == 6 ? ", \"unobservable\": \"no test can see it\"" : ""
~         let emitted = id == 8 ? ", \"emitted\": false" : ""
~         sites.append(
              $"{id > 1 ? "," : ""} {{ \"id\": {id}, \"file\": \"lib.byn\", \"line\": {id}, \"column\": 5, \"operator\": \"arithmetic\", \"original\": \"n + {id}\", \"mutated\": \"n - {id}\"{mark}{emitted} }}",
          )

~     goldenWrite(dir, for: "replay.sh", as: "case $BAYAN_MUTANT in\n  2|5|7) exit 1 ;;\n  4) sleep 7 ;;\nesac\nexit 0\n")

  // Criterion 3 and 4 of 019-mutspeed-a: the report, summary and site lines of --jobs 4 equal --jobs 1's, timing removed, the timeout too.
  test "mutate --binary: --jobs 1 and --jobs 4 print the same lines in site order and write the same report, timing removed" {
      if Target.os == "windows" {
          todo("the replay command is a `sh` script and the workers start through `sh`")
      }
      let dir = t13JobsFixture("mutate-jobs")
      let one = t13JobsRun(dir, by: ["--jobs", "1"], as: null)
      let four = t13JobsRun(dir, by: ["--jobs", "4"], as: null)
      let env = t13JobsRun(dir, by: List<string>(), as: "4")
      deleteWorkDir(dir)
      assert(one.exitCode == 0 && four.exitCode == 0 && env.exitCode == 0)
      assert(four.stdout == one.stdout && env.stdout == one.stdout)
      let lines = t11ReadAllLines(one.stdout)
      let statuses = [
          "survived",
          "killed",
          "survived",
          "timeout (counted as killed)",
          "killed",
          "marked",
          "killed",
          "not emitted",
          "survived",
      ]
      for id in 1..10 {
          assert(lines[id - 1] == $"lib.byn:{id}:5 arithmetic `n + {id}` -> `n - {id}`: {statuses[id - 1]}")
      }
      assert(lines[9] == "mutation: 3 killed, 3 survived, 1 timeout, 1 marked, 0 not reached, 1 not emitted")
  }
