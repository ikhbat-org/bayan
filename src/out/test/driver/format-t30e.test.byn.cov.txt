          build.append(a)
      }

      assert(o.exitCode == 0)

  /// FormatTests.cs:887 `FixKeepsTheSameC`; the check that `fix` changed the file is not in the C#: without it a fix that does nothing passes.
  @cases("style-literals-empty", "style-sections-banner")
  test "format t30e: FixKeepsTheSameC" (string name) {
      // FormatTests.cs:820 `FixCase(name)`
      let dir = f30eFresh($"fix-{name}")
      let file = Path.combine(dir, "main.byn")
      let input = Io.readText(Path.fullPath($"tests/cases/diag/{name}.byn")) catch ""
      assert(input != "")
      goldenWrite(dir, for: "main.byn", as: input)
      var emit = List<string>()
      for a in ["build", file, "--emit-c"] {
          emit.append(a)
      }
      let before = f30eRun(emit)
      print(before.stderr) if before.exitCode != 0
      assert(before.exitCode == 0)
      var fix = List<string>()
      fix.append("fix")
      fix.append(dir)
      assert(f30eRun(fix).exitCode == 0)
      assert((Io.readText(file) catch "") != input)
      let after = f30eRun(emit)
      print(after.stderr) if after.exitCode != 0
      assert(after.exitCode == 0)
      assert(before.stdout == after.stdout)
      deleteWorkDir(dir)
  }
