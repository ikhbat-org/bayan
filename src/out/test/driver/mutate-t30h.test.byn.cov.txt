  test "mutate t30h: SelectingAMutantRunsIt" (
      string op,
      string text,
      string original,
      string mutated,
      string plain,
      string expected
  ) {
      let name = op == "emptyReturn" ? "select-empty-return" : $"select-{op}"
      let root = t30hSchemata(name, by: op == "variant" ? t30hVariants() : text)
      let exe = t30hExe(root, for: name)
      let map = externalSiteReadAll(Io.readText(emitter.sitesPath(exe)) catch "") ?? panic("not a site map")
      var ids = List<i64>()
      for s in map {
          if s.op == op && s.original == original && s.mutated == mutated {
              ids.append(s.id)
              assert(s.emitted)
          }
      }
      assert(ids.length == 1)
      let plainOut = plain.length == 0 ? "" : $"{plain}\n"
      let expectedOut = expected.length == 0 ? "" : $"{expected}\n"
      assert(t30hRun(exe, by: null) == plainOut)
      assert(t30hRun(exe, by: $"{ids[0]}") == expectedOut)
      deleteWorkDir(root)
  }
  
  // MutantSchemataTests.cs:266 `ArmsWithPreludesBranch`: a switch is a C conditional only when no arm has a prelude; otherwise each
  // arm's prelude runs inside its own branch.
  test "mutate t30h: ArmsWithPreludesBranch" {
      let host = t5HostTarget()
      let plainRoot = t30hSchemata("arms-plain", by: "i32 f(i32 a) {\n    return a + 1\n}\n\nprint($\"{f(2)}\")\n")
      let plain = Io.readText(Path.combine(plainRoot, $"out/safe/{host}/c/main.c")) catch ""
      assert(plain.contains("return (byn_mutant_is(1) ? "))
      assert(!plain.contains("if (byn_mutant_is"))
      deleteWorkDir(plainRoot)
      // `try` lowers to a failing call before the `+`: the `-` mutant gets its own call in its own branch.
      let root = t30hSchemata(
          "arms-prelude",
          by: t30hVariants().replaced("return n", by: "return (try check(n - 1)) + 1"),
      )
      let prelude = Io.readText(Path.combine(root, $"out/safe/{host}/c/main.c")) catch ""
      let map = externalSiteReadAll(Io.readText(emitter.sitesPath(t30hExe(root, for: "arms-prelude"))) catch "")
          ?? panic("not a site map")
      var ids = List<i64>()
      for s in map {
          if s.op == "arithmetic" && s.original == "(try check(n - 1)) + 1" {
              ids.append(s.id)
          }
      }
      assert(ids.length == 1)
      let branch = prelude.position($"if (byn_mutant_is({ids[0]})) {{") ?? -1
      print(prelude) if branch < 0
      assert(branch >= 0)
      let end = prelude.position("\n}\n", from: branch) ?? prelude.length
      let body = prelude[branch..end]
      assert(body.count("= byn_main__check(") == 2)
      deleteWorkDir(root)
  }
