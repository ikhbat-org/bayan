  // L025 (spec 019 T31c, share S4): SelfhostEmitterNoPanicLawTests.cs without .NET. The corpora are written under tmp/l025-nopanic/ and each
  // is run through `bayan-emit --mode m --many` in the three modes, the stage-1 emitter program `goldenProgram` builds. The seed programs and the
  // corruption step are L023's (`l023SeedPrograms`, `l023Mutated`), so the two laws share one generator as the C# did (Q826); the seeds are the
  // 8 fixed ones only (the C# 256/1,024-seed tiers are not ported), and the tiers differ only in the corrupted count.
  
  /// SelfhostEmitterNoPanicLawTests.cs:23 `RunBoundMs`: the bound on one `bayan-emit --many` run (a mode and a corpus), in milliseconds.
  const l025RunBoundMs: i64 = 120000
  
  /// Writes `bytes[i]` as `names[i]` to the fresh directory `dir`: empty, or what failed.
  string l025Write(string dir, by List<string> names, from List<List<u8>> bytes) uses alloc, fs {
~     deleteWorkDir(dir)
~     Io.createDirectory(dir) catch return $"cannot create {dir}"
~     for i in 0..names.length {
~         Io.writeFile(Path.combine(dir, names[i]), bytes[i].span) catch return $"cannot write {dir}/{names[i]}"
      }
~     return ""
  }
  
  /// One `bayan-emit --mode mode --many dir` over the `*.byn` inputs `names` (sorted), as `RunBounded` and `RunCorpus` (:41-89) check it: empty
  /// when it exits 0 within `l025RunBoundMs` with no `panic:` on stderr, a section per input in order and only `exit 1` lines; else the
  /// counterexample. `firsts` gets each section's first line (empty for an empty section), the stand-in for the C# checker's verdict.
  string l025Many(
      string emit,
      at string dir,
      by List<string> names,
      for string corpus,
      as string mode,
      through ref List<string> firsts
  )
          uses alloc, clock, console, fs, process {
~     var failure = ""
~     var args = List<string>()
~     for a in ["--mode", mode, "--many", dir] {
~         args.append(a)
      }
~     let start = Io.monotonicNanos()
      // `whole` keeps every byte of stdout (a `timeout:` run keeps 64 KiB of each end), so the bound is measured, not a kill
~     let o = goldenChild(emit, by: args, at: dir, from: List<u8>(), through: ref failure, whole: true)
~     let ms = (Io.monotonicNanos() - start) / 1000000
~     return $"L025 counterexample: {corpus} ({mode}): {failure}" if failure != ""
~     firsts = List<string>()
~     var last = "(none)"
~     var next: i64 = 0
~     var first = false
~     for line in goldenLines(o.stdout) {
~         if next < names.length && line == $"== {names[next]}" {
~             last = $"{names[next]}"
~             firsts.append("")
~             first = true
~             next += 1
~             continue
          }
~         if first {
~             firsts[next - 1] = $"{line}"
~             first = false
          }
~         if line.startsWith("exit ") && line != "exit 1" {
~             return $"L025 counterexample: {corpus} ({mode}): {line} after {last} in {dir}"
          }
      }
~     if o.stderr.contains("panic:") || o.exitCode != 0 {
          // a crash loses the buffered sections, so the input is the first from `last` on that dies alone in a scratch directory
~         let alone = $"{dir}-alone"
~         for i in (next > 0 ? next - 1 : 0)..names.length {
~             deleteWorkDir(alone)
~             let bytes = Io.readFile(Path.combine(dir, names[i])) catch List<u8>()
~             Io.createDirectory(alone) catch {}
~             Io.writeFile(Path.combine(alone, names[i]), bytes.span) catch {}
~             var one = List<string>()
~             for a in ["--mode", mode] {
~                 one.append(a)
              }
~             if names[i].endsWith(".test.byn") {
~                 one.append("--tests")
              }
~             for a in ["--package", alone, Path.combine(alone, names[i])] {
~                 one.append(a)
              }
~             let r = goldenChild(emit, by: one, at: alone, from: List<u8>(), through: ref failure, whole: true)
~             if r.stderr.contains("panic:") || (r.exitCode != 0 && r.exitCode != 1) {
~                 return $"L025 counterexample: {corpus} ({mode}): exit {r.exitCode} on {names[i]} in {dir}, bytes {Hex.encode(bytes.span)}\n{r.stderr}"
              }
          }
~         return $"L025 counterexample: {corpus} ({mode}): exit {o.exitCode} after {last} in {dir}\n{o.stderr}"
      }
~     return $"L025 counterexample: {corpus} ({mode}): no section for {names[next]} in {dir}" if next < names.length
~     return $"L025 counterexample: {corpus} ({mode}): bayan-emit --many took {ms} ms, past {l025RunBoundMs} ms" if ms > l025RunBoundMs
~     print($"L025: {corpus} ({mode}): {names.length} inputs in {ms} ms")
~     return ""
  }
  
  // L025: the emitter never panics and exits 0 or 1 on every input in every mode
  test "L025: bayan-emit --many in debug, safe and fast over the 79 seed programs, their corrupted copies and the run inputs exits 0 and never panics" {
      let start = Io.monotonicNanos()
      var failure = ""
      let exe = goldenExe(through: ref failure)
      assert(failure == "")
      let emit = goldenProgram("emitter", by: exe, through: ref failure)
      print(failure) if emit == ""
      assert(emit != "")
      let full = (Env.get("BAYAN_LAW_TIER") ?? "") == "full"
      let size: i64 = full ? 10000 : 1000
      // SelfhostEmitterNoPanicLawTests.cs:27 `FloorQuick`, `FloorFull`: corrupted inputs that reach emission (their section has no `exit 1`)
      let floor: i64 = full ? 500 : 45
      let root = Path.combine(Path.fullPath("tmp"), "l025-nopanic")
      let modes: List<string> = ["debug", "safe", "fast"]
      var problem = ""
  
      let seeds = l023SeedPrograms()
      // the bytes L023 pins to the C# `PinnedCorpus`
      assert(seeds.length == 79)
      assert(l023SeedsHash(seeds) == "a38f8e5995f343c0eedf3062fe05e6119377dcf5114301df67504a53436290df")
      var seedNames = List<string>()
      var seedBytes = List<List<u8>>()
      for p in seeds {
          seedNames.append($"{p.name}")
          seedBytes.append(p.source.bytes.copy())
      }
      // the C# `checkRejected` oracle, without .NET: a `fault-*` program's section starts with `exit 1` and a `valid-*` one's does not
      let seedsDir = Path.combine(root, "seeds")
      problem = l025Write(seedsDir, by: seedNames, from: seedBytes)
      seedNames.sort()
      for mode in modes {
          break if problem != ""
          var firsts = List<string>()
          problem = l025Many(emit, at: seedsDir, by: seedNames, for: "seeds", as: mode, through: ref firsts)
          for i in 0..firsts.length {
              break if problem != ""
              if seedNames[i].startsWith("fault-") != (firsts[i] == "exit 1") {
                  problem = $"L025 counterexample: seeds/{seedNames[i]} ({mode}): its section starts with `{firsts[i]}`"
              }
          }
      }
  
      // SelfhostEmitterNoPanicLawTests.cs:33 `Corrupted`: the first `size` inputs of L023's random corpus over the 79 programs, seed 0x9E3779B97F4A7C15
      let corrupted = l023Mutated(seedBytes, by: 0x9E3779B97F4A7C15, for: size)
      var corruptedNames = List<string>()
      for i in 0..size {
          corruptedNames.append($"{l023Padded(i, to: 5)}.byn")
      }
      let corruptedDir = Path.combine(root, "corrupted")
      if problem == "" {
          problem = l025Write(corruptedDir, by: corruptedNames, from: corrupted)
      }
      var reaching: i64 = 0
      for mode in modes {
          break if problem != ""
          var firsts = List<string>()
          problem = l025Many(emit, at: corruptedDir, by: corruptedNames, for: "corrupted", as: mode, through: ref firsts)
          reaching = 0
          for f in firsts {
              reaching += f == "exit 1" ? 0 : 1
          }
          print($"L025: corrupted ({mode}): {reaching} of {size} inputs reach emission (floor {floor})")
          if problem == "" && reaching < floor {
              problem = $"L025: corrupted ({mode}): only {reaching} of {size} inputs reach emission, floor {floor}; the corruption no longer exercises the emitter"
          }
      }
  
      // the flat `tests/cases/run/*.byn` inputs, in byte order
      var runNames = List<string>()
      var runBytes = List<List<u8>>()
      let runCases = Path.fullPath("tests/cases/run")
      var listed = Io.listDir(runCases) catch List<string>()
      listed.sort()
      for name in listed {
          let path = Path.combine(runCases, name)
          continue if !name.endsWith(".byn") || !Fs.fileExists(path) || t6DirectoryExists(path)
          runNames.append($"{name}")
          runBytes.append(Io.readFile(path) catch List<u8>())
      }
      let runDir = Path.combine(root, "run")
      if problem == "" {
          problem = l025Write(runDir, by: runNames, from: runBytes)
      }
      for mode in modes {
          break if problem != ""
          var firsts = List<string>()
          problem = l025Many(emit, at: runDir, by: runNames, for: "run", as: mode, through: ref firsts)
      }
  
      print(
          $"L025: seeds {seeds.length}, corrupted {size} ({reaching} reach emission), run {runNames.length}, {(Io.monotonicNanos() - start) / 1000000} ms",
      )
      print(problem) if problem != ""
      assert(problem == "")
      deleteWorkDir(root)
  }
