  // 019 T29 (coverage ratchet): binder3.byn paths no golden case reaches. Each program is compiled alone as `a.byn` and
  // summarised as its diagnostics (`code start+length message`, joined by ` | `); the expected text is the output of the
  // self-hosted checker at the commit that added the test.
  
  import lexer
  
  string t29b3Diags(string text, bool library = false) uses alloc, clock, fs {
~     var bytes = List<u8>()
~     for b in text.bytes {
~         bytes.append(b)
      }
~     var sources = List<lexer.SourceText>()
~     sources.append(lexer.SourceText.create("a.byn", from: move bytes))
~     let c = create(from: move sources, at: ".", by: CompilationOptions { library: library })
~     var sb = StrBuf()
~     var first = true
~     for d in c.diagnostics {
~         sb.append(" | ") unless first
~         first = false
~         sb.append($"{d.code} {d.span.start}+{d.span.length} {d.message}")
      }
~     return $"{sb.finish()}"
  }
  
  bool t29b3Is(string text, as string expected, bool library = false) uses alloc, clock, console, fs {
~     let got = t29b3Diags(text, library: library)
~     if got != expected {
~         print($"T29GOT {got}")
      }
~     return got == expected
  }

  test "binder3 t29: for over generic, enumerator and fallible iterables" {
      let text = """
              union error ReadError { broken }
  
              struct Rows {
                  i64 at
  
                  i64? next(ref this) fails ReadError {
                      at += 1
                      return at < 3 ? at : null
                  }
              }
  
              struct NoNext {
                  i64 at
  
                  i64? next(ref this, i64 by) fails ReadError {
                      return null
                  }
              }
  
              interface IMark {
                  i64 mark()
              }
  
              struct Countdown : IMark, IEnumerator {
                  type Item = i64
  
                  i64 left
  
                  i64 mark() => 1
  
                  bool moveNext(ref this) {
                      left -= 1
                      return left >= 0
                  }
  
                  i64 current => left
              }
  
              struct NoItem : IEnumerator {
                  i64 left
  
                  bool moveNext(ref this) => false
  
                  i64 current => left
              }
  
              Rows makeRows() => Rows { at: 0 }
  
              NoNext noNext() => NoNext { at: 0 }
  
              NoItem noItem() => NoItem { left: 1 }
  
              Countdown from(i64 n) => Countdown { left: n }
  
              void loose<T>(T xs) {
                  for x in xs {
                      print("x")
                  }
              }
  
              void steps<T>(T xs) where T : IEnumerator {
                  for x in xs {
                      print("x")
                  }
              }
  
              T same<T>(T x) => x
  
              void steps2<T>(T xs) where T : IEnumerator {
                  for x in same(xs) {
                      print("x")
                  }
              }
  
              void main2(List<i64> xs) fails ReadError {
                  var rows = Rows { at: 0 }
                  for try r: string in rows {
                      print("r")
                  }
                  for try r in (rows) {
                      print("r")
                  }
                  for try r in makeRows() {
                      print("r")
                  }
                  for try r in noNext() {
                      print("r")
                  }
                  for n: string in xs {
                      print("n")
                  }
                  for n in from(2) {
                      print($"{n}")
                  }
                  for n in noItem() {
                      print($"{n}")
                  }
              }
              """
      assert t29b3Is(
          text,
          as: "BYN2803 523+11 `NoItem` does not bind `type Item` of `IEnumerator` | BYN2221 810+2 `T` cannot be iterated | BYN2202 1139+6 expected `string`, found `i64` | BYN2221 1304+8 `NoNext` cannot be iterated | BYN2202 1351+6 expected `string`, found `i64` | BYN2104 900+2 `xs` is immutable | BYN2780 951+1 implicit copy of a unique value: type parameter `T` may stand for a unique type",
      )
  }
