  // 019 T29 (coverage ratchet): bound.byn paths no golden case reaches. The node accessors are read back from nodes added
  // to the context of an empty program; the expected text is the output of these functions at the commit that added the
  // test.
  
  import lexer
  
  Ctx t29BoundCtx() uses alloc, clock, fs {
~     var sources = List<lexer.SourceText>()
~     sources.append(lexer.SourceText.create("a.byn", from: List<u8>()))
~     return create(from: move sources, at: ".", by: CompilationOptions {})
  }
  
  TextSpan t29bdAt(i64 start) => TextSpan { start: start, length: 1 }
  
  bool t29BoundIs(string got, as string expected) uses alloc, console {
~     if got != expected {
~         print($"T29GOT {got}")
      }
~     return got == expected
  }
  
  /// `start` of `nodeSpan` and the display of `exprType` (or `patternInput` when `pattern`) of `n`.
  string t29bdNode(ref Ctx c, for BNode n, bool pattern = false) uses alloc {
~     let type = pattern ? patternInput(in c, for: n) : exprType(in c, for: n)
~     return $"{nodeSpan(in c, for: n).start} {typeDisplay(ref c, by: type)}"
  }
  
  test "bound t29: constantText shows every kind" {
      assert t29BoundIs(constantText(Constant { kind: .integer, integer: bigFromI64(-12) }), as: "-12")
      assert t29BoundIs(constantText(Constant { kind: .real, real: 2.5 }), as: "2.5")
      assert t29BoundIs(constantText(Constant { kind: .flag, flag: true }), as: "true")
      assert t29BoundIs(constantText(Constant { kind: .flag, flag: false }), as: "false")
      assert t29BoundIs(constantText(Constant { kind: .text, text: "hi" }), as: "\"hi\"")
      assert t29BoundIs(constantText(Constant { kind: .rune, rune: 0x41 }), as: "A")
  }
  
  /// A runtime division, so that `0.0 / 0.0` and `1.0 / 0.0` are not folded.
  f64 t29BoundQuotient(f64 a, by f64 b) => a / b
  
  test "bound t29: realText names the non-finite values" {
      assert t29BoundIs(realText(t29BoundQuotient(0.0, by: 0.0)), as: "NaN")
      assert t29BoundIs(realText(t29BoundQuotient(1.0, by: 0.0)), as: "Infinity")
      assert t29BoundIs(realText(t29BoundQuotient(-1.0, by: 0.0)), as: "-Infinity")
  }

  test "bound t29: the input type of every pattern" {
      var c = t29BoundCtx()
      let constant = addConstantPattern(
          ref c,
          from: BConstantPattern { span: t29bdAt(10), inputType: prim(.i8), value: -1 },

      let relational = addRelationalPattern(

      assert t29BoundIs(t29bdNode(ref c, for: relational, pattern: true), as: "11 i16")
      let range = addRangePattern(
          ref c,

  test "bound t29: syntax records skip -1, share a node's syntax and type a state literal" {
      var c = t29BoundCtx()
      let plain = addSpread(ref c, from: BSpread { span: t29bdAt(1), operand: -1, type: prim(.i64) })
      assert recordSyntaxNode(ref c, for: plain, at: -1) == plain
      assert c.exprSyntaxOf.tryGet(at: plain) is null
      let block = addSpread(ref c, from: BSpread { span: t29bdAt(2), operand: -1, type: prim(.i64) })
      let _ = recordNodeSyntax(ref c, for: block, as: .blockStmt, at: t29bdAt(2))
      let shared = shareSyntax(
          ref c,
          for: addSpread(ref c, from: BSpread { span: t29bdAt(3), operand: -1, type: prim(.i64) }),
          from: block,
      )
      assert (c.nodeSyntaxOf.tryGet(at: shared) ?? -1) == (c.nodeSyntaxOf.tryGet(at: block) ?? -2)
      let state = addStructLiteral(ref c, from: BStructLiteral { span: t29bdAt(4), structId: 0, state: 1 })
      assert kindOf(exprType(in c, for: state)) == .structType
  }
