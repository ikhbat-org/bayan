  // 019 T29 (coverage ratchet): express.byn paths no golden case reaches. Each program is compiled alone as `a.byn` and
  // summarised as its diagnostics (`code start+length message`, joined by ` | `); the expected text is the output of the
  // self-hosted checker at the commit that added the test.
  
  import lexer
  
  string t29exDiags(string text, string other = "") uses alloc, clock, fs {
~     var bytes = List<u8>()
~     for b in text.bytes {
~         bytes.append(b)
      }
~     var sources = List<lexer.SourceText>()
~     sources.append(lexer.SourceText.create("a.byn", from: move bytes))
~     if other != "" {
~         var otherBytes = List<u8>()
~         for b in other.bytes {
~             otherBytes.append(b)
          }
~         sources.append(lexer.SourceText.create("util/u.byn", from: move otherBytes))
      }
~     let c = create(from: move sources, at: ".", by: CompilationOptions { library: true })
~     var sb = StrBuf()
~     var first = true
~     for d in c.diagnostics {
~         sb.append(" | ") unless first
~         first = false
~         sb.append($"{d.code} {d.span.start}+{d.span.length} {d.message}")
      }
~     return $"{sb.finish()}"
  }
  
  bool t29exIs(string text, as string expected, string other = "") uses alloc, clock, console, fs {
~     let got = t29exDiags(text, other: other)
~     if got != expected {
~         print($"T29GOT {got}")
      }
~     return got == expected
  }
  
  bool t29exTextIs(string got, as string expected) uses console {
~     if got != expected {
~         print($"T29GOT {got}")
      }
~     return got == expected
  }
  
  bool t29exF64Is(f64 got, as string expected) uses alloc, console => t29exTextIs($"{got}", as: expected)

  test "express t29: functions passed where a function type is expected" {
      let text = """

                  bad
              }
  
              union error F {
                  worse
              }
  
              union Shape {
                  rect(i64 w, i64 h)
                  dot
              }
  
              struct P {
                  i64 x
                  i64 add(i64 a, to i64 b) => a + b
              }
  
              i64 mods(ref i64 a, from move string s, to out i64 o, at in i64 q) {
                  o = 0
                  return a
              }
  
              void loose() fails {
                  return
              }
  
              T id<T>(T x) => x
  
              i64 apply(i64 -> i64 f) => f(1)
  
              void run(() -> void fails E, F f) fails E, F {
                  try f()
              }

              void main() {
                  let a = apply(mods)
                  run(loose)

  test "express t29: type arguments on a name that is not a generic function" {
      let head = """
              interface IFoo {
                  i64 n()
              }
  
              union error E {
                  bad
              }
  
              const n: i64 = 4
  
              struct G<T> {
                  i64 v
              }
  
              void main() {
                  let a = G<
              """
      assert t29exIs(
          $"{head}(i64, string) -> i64 uses alloc fails E>()\n}}\n",
          as: "BYN2822 132+39 a function type is not supported as a type argument yet",
      )
      assert t29exIs(
          $"{head}(i64, string) -> i64 uses alloc fails E>\n}}\n",
          as: "BYN2822 132+39 a function type is not supported as a type argument yet",
      )
      assert t29exIs($"{head}dyn ref IFoo>()\n}}\n", as: "BYN2109 130+17 `G` needs a value for `v`")
      assert t29exIs($"{head}dyn ref IFoo>\n}}\n", as: "BYN2019 130+15 `G<dyn ref IFoo>` is a type, not a value")
      assert t29exIs($"{head}i64[4]>()\n}}\n", as: "BYN2109 130+11 `G` needs a value for `v`")
      assert t29exIs($"{head}i64[4]>\n}}\n", as: "BYN2019 130+9 `G<i64[4]>` is a type, not a value")
      assert t29exIs($"{head}i64[n]>()\n}}\n", as: "BYN2109 130+11 `G` needs a value for `v`")
      assert t29exIs($"{head}i64[n]>\n}}\n", as: "BYN2019 130+9 `G<i64[4]>` is a type, not a value")
      assert t29exIs($"{head}(i64, i64)>()\n}}\n", as: "BYN2109 130+15 `G` needs a value for `v`")
      assert t29exIs($"{head}(i64, i64)>\n}}\n", as: "BYN2019 130+13 `G<(i64, i64)>` is a type, not a value")
      assert t29exIs($"{head}i64?>()\n}}\n", as: "BYN2109 130+9 `G` needs a value for `v`")
      assert t29exIs($"{head}i64?>\n}}\n", as: "BYN2019 130+7 `G<i64?>` is a type, not a value")
      assert t29exIs(
          $"{head}i64 -> i64>\n}}\n",
          as: "BYN2822 132+10 a function type is not supported as a type argument yet",
      )
  }
