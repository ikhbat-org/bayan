  // 019 T29 (coverage ratchet): conformance.byn's signature texts, fixes, associated types and generic defaults, which no
  // golden case reaches. The expected text is the output of the self-hosted code at the commit that added the test.
  
  /// The diagnostics of `text` compiled as `a.byn`, each as `code message => fix;` (the fix's replacement, when it has one).
  string t29s0Fixes(string text) uses alloc, clock, fs {
~     let c = t29s0Compile(text)
~     var sb = StrBuf()
~     for d in c.diagnostics {
~         sb.append($"{d.code} {d.message}")
~         if d.hasFix {
~             sb.append($" => {d.fixSpan.start}+{d.fixSpan.length} {d.fixReplacement}")
          }
~         sb.append(";")
      }
~     return $"{sb.finish()}"
  }
  
  test "conformance t29: missing members show their parameter modifiers and `fails`" {
      let text = "union error E { bad }\n\ninterface IAll {\n    void m(ref i64 a, move string b, out i64 c, in string d, i64 e)\n    void f() fails\n    void g() fails E\n    i64 count\n}\n\nstruct S : IAll { }\n"
      assert t29s0Same(
          t29s0Fixes(text),
          as: "BYN5004 `m` has 5 unlabeled parameters; give each but one a particle;BYN5006 `b` is a required parameter besides the object and needs a particle;BYN5006 `c` is a required parameter besides the object and needs a particle;BYN5006 `d` is a required parameter besides the object and needs a particle;BYN5006 `e` is a required parameter besides the object and needs a particle;BYN2801 `S` does not provide `m` required by `IAll` => 183+0 void m(ref i64 a, move string b, out i64 c, in string d, i64 e) => todo(\"m\") ;BYN2801 `S` does not provide `f` required by `IAll` => 183+0 void f() fails => todo(\"f\") ;BYN2801 `S` does not provide `g` required by `IAll` => 183+0 void g() fails E => todo(\"g\") ;BYN2801 `S` does not provide `count` required by `IAll` => 183+0 i64 count => todo(\"count\") ;",
      )
  }
