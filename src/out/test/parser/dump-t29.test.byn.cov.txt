  // 019 T29 (coverage ratchet): dump.byn paths no golden case reaches: the `unique` and `unsafe` modifiers, a bracket
  // attribute, `become` with a named argument, attributes on an empty and an error statement, and a unit whose only
  // diagnostic has no code. The helper `t3ParserOf` is recover.test.byn's. The expected texts are the output of the
  // self-hosted parser at the commit that added the test.
  
  /// The dump of `text` parsed as a compilation unit.
  string t29DumpOf(string text) uses alloc {
~     var p = t3ParserOf(text)
~     let unit = parseCompilationUnit(ref p)
~     var sb = StrBuf()
~     dump(in unit, from: ref p, to: ref sb)
~     return $"{sb.finish()}"
  }
  
  /// The tree of `text` parsed as a compilation unit, one line per declaration and statement, then a
  /// `[code start+length]` line per diagnostic; the short form the other `-t29` tests compare.
  string t29Brief(string text) uses alloc {
~     var p = t3ParserOf(text)
~     let unit = parseCompilationUnit(ref p)
~     var sb = StrBuf()
~     for member in unit.members {
~         decl(in member, at: 0, from: ref p, to: ref sb)
      }
~     for d in p.lexed.diagnostics {
~         sb.append($"[{d.code} {d.span.start}+{d.span.length}]\n")
      }
~     return $"{sb.finish()}"
  }
  
  /// `t29Brief` of each body in `bodies`, each inside `void f() { … }`, after a `== <index>` line.
  string t29Bodies(ReadOnlySpan<string> bodies) uses alloc {
~     var sb = StrBuf()
~     for i in 0..bodies.length {
~         sb.append($"== {i}\n")
~         sb.append(t29Brief($"void f() {{\n{bodies[i]}\n}}\n"))
      }
~     return $"{sb.finish()}"
  }
  
  test "parser t29: the dump writes rare modifiers and statement forms" {
      var sb = StrBuf()
      sb.append($"[{mods(modifierUnique | modifierUnsafe | modifierExtern)}]\n")
      sb.append(t29DumpOf("[Obsolete] void f() { }\n"))
      sb.append(t29DumpOf("void f() {\n    become .retry(n: 1, 2)\n    @a ;\n    @a )\n}\n"))
      sb.append(t29DumpOf("void f() {\n    unless (!x) {\n    }\n    @a\n}\n"))
      let expected = """
              [unique extern unsafe ]
              CompilationUnit
                Attribute @obsolete
                Function f() -> void

              error BYN1045: `[Obsolete]` is an input form; write `@obsolete`

                  │ ^
                  │
                help: write the attribute with `@` and a camelCase name
                  │ 1 │ @obsolete void f() { }
                  │   │ +++++++++
              CompilationUnit
                Function f() -> void
                  Block
                    Become .retry n: 1 2
                    Attribute @a
                    Empty
                    Attribute @a
                    Expr <error>
  
              error BYN1069: attribute `@a` is not allowed before a statement
  
                3 │     @a ;
                  │     ^^ before a statement
                  │
                help: remove the attribute; only `@unobservable` may come before a statement
  
              error BYN1040: `;` is an input form; a statement ends at the newline
  
                3 │     @a ;
                  │        ^ remove `;`
                  │
                help: remove the `;`
                  │ 3 │     @a ;
                  │   │       --
  
              error BYN1069: attribute `@a` is not allowed before a statement
  
                4 │     @a )
                  │     ^^ before a statement
                  │
                help: remove the attribute; only `@unobservable` may come before a statement
  
              error BYN1002: expected an expression, found `)`
