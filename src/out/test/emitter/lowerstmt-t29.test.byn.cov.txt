  type LsBReturn = semantics.BReturn
  
  semantics.Ctx lsCompile(string source) {
~     var bytes = List<u8>()
~     for b in source.bytes {
~         bytes.append(b)
      }
~     return t12Compile(Path.fullPath("tmp"), at: move bytes, for: "ls.byn", by: false)
  }
  
  string lsEmit(string source, by bool covering, as bool mutating) {
~     var c = lsCompile(source)
~     var sb = StrBuf()
~     for d in c.diagnostics {
~         sb.append($"{d.code} {d.span.start} {d.message}\n") if !d.isWarning
      }
~     let errors = sb.finish()
~     return $"errors: {errors}" if errors != ""
~     return t10Emit(ref c, by: "safe", through: covering, as: mutating)
  }

  test "lowerstmt t29: compound assignment to an index that panics" {

  FuncId lsFn(semantics.Ctx c, for string name) {
~     for k in 0..c.functions.length {
~         return k if c.functions[k].name == name
      }
~     return -1
  }
  
  BExpr lsCall(ref semantics.Ctx c, for string name) {
~     let f = lsFn(c, for: name)
~     return semantics.addCall(
          ref c,
          from: BCall { span: TextSpan { start: 0, length: 1 }, function: f, type: c.functions[f].returnType },
      )
  }
  
  BExpr lsLit(ref semantics.Ctx c, as i64 v) {
~     return i64Literal(ref c, at: TextSpan { start: 0, length: 1 }, by: v)
  }
  
  test "lowerstmt t29: hand-built lvalues, returns and loops" {
      var c = lsCompile(
          """
          struct Pt {
              i64 x = 0
          }
  
          Pt mk() => Pt { x: 1 }
  
          (i64 a, i64 b) tup() => (a: 1, b: 2)
  
          i64[2] arr() => [1, 2]
  
          List<i64> list() => [1, 2]
  
          Map<i64, i64> mp() => Map<i64, i64>()
  
          i64 one() => 1
  
          void v() {
          }
  
          struct Rows : IIndexable {
              type Item = i64
  
              i64 count => 2
  
              i64 get(i64 index) => index
          }
  
          Rows rows() => Rows {}
  
          void main() {
              for x in rows() {
              }
          }
          """,
      )
      assert c.diagnostics.length == 0
      var l = LCtx {}
      l.fn = lsFn(c, for: "main")
      l.outs.append(List<BStmt>())
      let span = TextSpan { start: 0, length: 1 }
      let i64t = semantics.prim(.i64)
      var x = -1
      for k in 0..c.fields.length {
          x = k if c.fields[k].name == "x"
      }
      let field = semantics.addField(
          ref c,
          from: BField { span: span, receiver: lsCall(ref c, for: "mk"), field: x, type: i64t },
      )
      let f1 = lowerLvalue(ref c, through: ref l, for: field)
      let tuple = semantics.addTupleElement(
          ref c,
          from: BTupleElement { span: span, receiver: lsCall(ref c, for: "tup"), index: 1, type: i64t },
      )
      let f2 = lowerLvalue(ref c, through: ref l, for: tuple)
      let index = semantics.addIndex(
          ref c,
          from: BIndex { span: span, target: lsCall(ref c, for: "arr"), index: lsLit(ref c, as: 1), type: i64t },
      )
      let f3 = lowerLvalue(ref c, through: ref l, for: index)
      let listLocal = newTemp(ref c, through: ref l, for: c.functions[lsFn(c, for: "list")].returnType)
      let mapLocal = newTemp(ref c, through: ref l, for: c.functions[lsFn(c, for: "mp")].returnType)
      var panicArguments = List<BExpr>()
      panicArguments.append(
          semantics.addLiteral(
              ref c,
              from: BLiteral { span: span, type: semantics.prim(.string), value: Constant { kind: .text, text: "p" } },
          ),
      )
      let panicCall = semantics.addCall(
          ref c,
          from: BCall {
              span: span,
              function: c.builtinPanic,
              arguments: move panicArguments,
              type: semantics.prim(.neverKind),
          },
      )
      let neverIndex = semantics.addIndex(
          ref c,
          from: BIndex { span: span, target: localExpr(ref c, at: span, for: listLocal), index: panicCall, type: i64t },
      )
      let f4 = lowerLvalue(ref c, through: ref l, for: neverIndex)
      var slotArguments = List<BExpr>()
      slotArguments.append(lsLit(ref c, as: 3))
      let slot = semantics.addBuiltinCall(
          ref c,
          from: BBuiltinCall {
              span: span,
              member: .mapSlot,
              receiver: localExpr(ref c, at: span, for: mapLocal),
              arguments: move slotArguments,
              type: i64t,
          },
      )
      let f5 = lowerLvalue(ref c, through: ref l, for: slot)
      let length = semantics.addBuiltinCall(
          ref c,
          from: BBuiltinCall {
              span: span,
              member: .listLength,
              receiver: localExpr(ref c, at: span, for: listLocal),
              type: i64t,
          },
      )
      let f6 = lowerLvalue(ref c, through: ref l, for: length)
      let localTuple = semantics.addTupleElement(
          ref c,
          from: BTupleElement { span: span, receiver: localExpr(ref c, at: span, for: listLocal), index: 0, type: i64t },
      )
      assert isLvalueShape(ref c, through: ref l, for: localTuple) == true
      let void = ltree.addLVoid(ref l.tree, from: T5LVoid { span: span })
      assert isLvalueShape(ref c, through: ref l, for: void) == false
      var setArguments = List<BExpr>()
      setArguments.append(lsCall(ref c, for: "one"))
      setArguments.append(lsLit(ref c, as: 4))
      let set = semantics.addBuiltinCall(
          ref c,
          from: BBuiltinCall {
              span: span,
              member: .mapIndexSet,
              receiver: localExpr(ref c, at: span, for: mapLocal),
              arguments: move setArguments,
              type: i64t,
          },
      )
      let assign = semantics.addAssign(
          ref c,
          from: BAssign { span: span, target: set, value: lsLit(ref c, as: 5), op: .add },
      )
      lowerStatement(ref c, through: ref l, for: assign)
      var userFor = -1
      for k in 0..c.bFors.length {
          userFor = k if c.bFors[k].userIteration is not null
      }
      c.bFors[userFor].iterable = semantics.addDefault(
          ref c,
          from: BDefault { span: span, type: c.functions[lsFn(c, for: "rows")].returnType },
      )
      let before = l.outs[0].length
      lowerStatement(
          ref c,
          through: ref l,
          for: semantics.addReturn(ref c, from: LsBReturn { span: span, value: lsCall(ref c, for: "v") }),
      )
      let afterReturn = l.outs[0].length
      let returned = l.outs[0][afterReturn - 1]
      assert ltree.lKindOf(returned) == .returnStmt
      assert l.tree.returnStmts[ltree.lIndexOf(returned)].value == -1
      let bare = semantics.addBuiltinCall(
          ref c,
          from: BBuiltinCall {
              span: span,
              member: .mapIndexSet,
              receiver: localExpr(ref c, at: span, for: mapLocal),
              type: i64t,
          },
      )
      lowerStatement(
          ref c,
          through: ref l,
          for: semantics.addAssign(
              ref c,
              from: BAssign { span: span, target: bare, value: lsLit(ref c, as: 6), op: .add },
          ),
      )
      let afterBare = l.outs[0].length
      lowerStatement(ref c, through: ref l, for: semantics.makeBound(.forStmt, at: userFor))
      assert $"{semantics.boundKind(f1)} {semantics.boundKind(f2)} {semantics.boundKind(f3)} {ltree.isLowered(f4)} {semantics.boundKind(f5)} {semantics.boundKind(f6)} {before} {afterReturn} {afterBare} {l.outs[0].length}" == "field tupleElement index true builtinCall local 8 10 12 17"
  }
