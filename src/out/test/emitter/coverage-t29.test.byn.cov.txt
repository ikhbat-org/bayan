  // 019 T29 (coverage ratchet): `instrument` and a `--coverage` emit over the corpus of lower-t29.test.byn (guards, `do`,
  // `fail`, `try` and `catch` in failing functions and methods, an error union less a case), `skipFiles`, and the
  // counters of small programs: generic instances, `while true`, `catch` in a function that cannot fail, `fail e`,
  // `expects`, a struct rule after a field rule, and error sets that share a set. The
  // expected values are the output of the self-hosted emitter at the commit that added the test.
  
  test "emitter t29: instrument counts the corpus's lines, branches and error paths" {
      let corpus = t29Corpus()
      var failed = StrBuf()
      for k in 0..corpus.length {
          var c = t29Compile(corpus[k], by: true)
          let compiled = t29Errors(in c)
          if compiled != "" {
              failed.append($"{k}: {compiled}")
              continue
          }
          let map = instrument(ref c, for: in List<string>())
          var lines = 0
          var branches = 0
          var errors = StrBuf()
          for counter in map.counters {
              if counter.kind == "line" {
                  lines += 1
              } else if counter.kind == "branch" {
                  branches += 1
              } else {
                  errors.append(
                      $" {counter.function ?? ""}:{counter.variant ?? ""}:{counter.origin ?? ""}@{counter.line}",
                  )
              }
          }
          assert lines > 0
          if k == 0 {
              assert lines == 27 && branches == 10
              let expected = " pick:F.a:fail@30 pick:F.b:fail@31 narrow:F.a:fail@36 viaTry:F.a:try@41 viaTry:F.b:try@41 viaTry:F.a:try@42 viaTry:E.worse:fail@43 viaCatch:F.b:catch@48 Counter.bump:F.a:fail@17 Counter.twice:F.a:try@23 Counter.twice:F.b:try@23 Counter.twice:F.b:catch@24"
              assert errors.finish() == expected
          }
          var skip = List<string>()
          skip.append(c.files[0].path)
          assert instrument(ref c, for: in skip).counters.length == 0
          assert t29Emit(corpus[k], covering: true).contains("byn_cov[")
      }
      assert failed.finish() == ""
  }
  
  /// The counters `instrument` makes for `source`, one `kind line:column function variant origin outcome` each.
  string t29CoverageDump(string source) {
~     var c = t29Compile(source)
~     let errors = t29Errors(in c)
~     return $"errors: {errors}" if errors != ""
~     var sb = StrBuf()
~     for counter in instrument(ref c, for: in List<string>()).counters {
~         sb.append($"{counter.kind} {counter.line}:{counter.column}")
~         sb.append($" {counter.function ?? "-"} {counter.variant ?? "-"} {counter.origin ?? "-"}")
~         sb.append($" {counter.outcome}\n")
      }
~     return sb.finish()
  }
