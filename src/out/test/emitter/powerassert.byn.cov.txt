  /// `Bar`: the column bar.

  /// `Indent`: starts every diagram line, so the bars sit under the statement after the runtime's `panic: `.

  /// `Piece`: `text`, or the value of column `value` when that is not -1. It is written when every column in `allShown`
  /// is shown, none in `noneShown` is, and, when `anyFrom` is not -1, some column from that index on is (C#'s `When`,
  /// `Unless` and `AnyFrom`; `anyFrom` -1 is null).
  struct PowerAssertPiece {
      string    text
      i64       value
      List<i64> allShown
      List<i64> noneShown
      i64       anyFrom
  }
  
  /// One line of `Flatten`'s input with the blanks around its break cut off; `next` is where the next line starts, -1 for the last.
  struct PowerAssertLine {
      i64 from
      i64 until
      i64 next
  }
  
  /// PowerAssert.cs:16 `Flatten`'s trimming of the line of `text` that starts at `start`: `TrimStart(' ', '\t')` when it is not the first line
  /// and `TrimEnd(' ', '\t', '\r')` when it is not the last.
  PowerAssertLine powerAssertLine(string text, at i64 start, by bool first) {
~     var end = text.length
~     var next: i64 = -1
~     let at = text.position("\n", from: start) ?? -1
~     if at >= 0 {
~         end = at
~         next = at + 1
      }
~     var from = start
~     var until = end
~     if !first {
~         while from < until && (text.bytes[from] is 32 or 9) {
~             from += 1
          }
      }
~     if next >= 0 {
~         while until > from
                  && (text.bytes[until - 1] == 32 || text.bytes[until - 1] == 9 || text.bytes[until - 1] == 13) {
~             until -= 1
          }
      }
~     return PowerAssertLine { from: from, until: until, next: next }
  }
  
  /// PowerAssert.cs:16 `Flatten`: source text on one line; each line break, with the blanks around it, becomes one space.
  string powerAssertFlatten(string text) uses alloc {
~     unless text.contains("\n") {
~         return text
      }
~     var sb = StrBuf()
~     var start: i64 = 0
~     var first = true
~     while start >= 0 {
~         let line = powerAssertLine(text, at: start, by: first)
~         unless first {
~             sb.append(" ")
          }
~         sb.append(text[line.from..line.until])
~         first = false
~         start = line.next
      }
~     return $"{sb.finish()}"
  }
  
  /// PowerAssert.cs:29 `Column` (Q848): the rune count of `before` once flattened.
  public i64 powerAssertColumn(string before) {
~     unless before.contains("\n") {
~         return before.runeCount
      }
~     var count: i64 = 0
~     var start: i64 = 0
~     var first = true
~     while start >= 0 {
~         let line = powerAssertLine(before, at: start, by: first)
~         unless first {
~             count += 1
          }
~         count += before[line.from..line.until].runeCount
~         first = false
~         start = line.next
      }
~     return count

  /// PowerAssert.cs:47 `Diagram`'s local `Add`: appends a piece, dropping an `anyFrom` past the last column, and an `anyFrom`
  /// that an always-shown column from there on satisfies, and the columns of `whens` that are always shown.
  void powerAssertAdd(

      to ref List<PowerAssertPiece> pieces,
      for string text,
      at i64 value,

      from List<i64> unlesses,
      as i64 anyFrom
  ) uses alloc {
~     var from = anyFrom
~     if from != -1 {
~         return if from >= columns.length
~         for j in from..columns.length {
~             if always[j] {
~                 from = -1
~                 break
              }
          }
      }
~     var kept = List<i64>()
~     for k in whens {
~         unless always[k] {
~             kept.append(k)
          }
      }
~     var noneShown = List<i64>()
~     for k in unlesses {
~         noneShown.append(k)
      }
~     pieces.append(
          PowerAssertPiece { text: text, value: value, allShown: move kept, noneShown: move noneShown, anyFrom: from },
      )
  }
  
  /// PowerAssert.cs:47 `Diagram`'s local `Gap`.
  string powerAssertGap(List<i64> columns, at i64 k) uses alloc {

  /// PowerAssert.cs:47 `Diagram`: the pieces after the statement, for ascending, distinct `columns`: a row of bars, then
  /// each value hung from its bar, rightmost first, under the bars to its left. A column that is not `always` shown (a side

  List<PowerAssertPiece> powerAssertDiagram(List<i64> columns, by List<bool> always) uses alloc {

                  columns,
                  by: always,

                  through: none,
                  from: only,
                  as: k + 1,

          }
      }

~         var only = List<i64>()
~         only.append(k)

~         for j in 0..k {

~             both.append(j)

  /// PowerAssertTests.cs:11 `Render`: the text `pieces` write when the columns in `shown` are shown, a value as `<column>`.
  string powerAssertRender(List<PowerAssertPiece> pieces, by List<i64> columns, for List<bool> shown) uses alloc {
~     var sb = StrBuf()
~     for p in pieces {
~         var written = true
~         for k in p.allShown {
~             written = written && shown[k]
          }
~         for k in p.noneShown {
~             written = written && !shown[k]
          }
~         if p.anyFrom != -1 {
~             var any = false
~             for k in p.anyFrom..shown.length {
~                 any = any || shown[k]
              }
~             written = written && any
          }
~         continue unless written
~         sb.append(p.value < 0 ? p.text : $"<{columns[p.value]}>")
      }
~     return $"{sb.finish()}"
  }

  // PowerAssertTests.BarsSitUnderEachColumnAndValuesHangRightToLeft.
  test "power-assert: bars sit under each column and values hang right to left" {
      let i = powerAssertIndent
      assert(powerAssertAlways([2, 6, 9]) == $"\n{i}  │   │  │\n{i}  │   │  <9>\n{i}  │   <6>\n{i}  <2>")
      for p in powerAssertDiagram([2, 6, 9], by: [true, true, true]) {
          assert(p.allShown.length == 0 && p.noneShown.length == 0 && p.anyFrom == -1)
      }
  }
  
  // PowerAssertTests.AColumnThatDidNotRunIsLeftOut.
  test "power-assert: a column that did not run is left out" {
      let columns: List<i64> = [2, 6, 9, 12]
      let always: List<bool> = [true, false, true, false]
      let pieces = powerAssertDiagram(columns, by: always)
      for six in [false, true] {
          for twelve in [false, true] {
              var shown = List<i64>()
              for k in 0..columns.length {
                  if always[k] || (columns[k] == 6 ? six : twelve) {
                      shown.append(columns[k])
                  }
              }
              assert(powerAssertAlways(shown) == powerAssertRender(pieces, by: columns, for: [true, six, true, twelve]))
          }
      }
      for p in pieces {
          for k in p.allShown {
              assert(!always[k])
          }
          for k in p.noneShown {
              assert(!always[k])
          }
      }
  }
  
  // PowerAssertTests.NoTrailingBlankWhenTheLastColumnsDidNotRun.
  test "power-assert: no trailing blank when the last columns did not run" {

      assert(powerAssertRender(pieces, by: columns, for: [false, false]) == "")
