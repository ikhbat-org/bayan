  import semantics
  import emitter.ltree
  import lexer
  
  // T29: func.byn's statements and ownership checks on a bound program and a hand-built `LTree`; this module cannot
  // import the lowerer (emitter imports emitter.cprint), so each test builds the lowered nodes it emits.
  
  type FnOptions = semantics.CompilationOptions
  type FnTree    = ltree.LTree
  type FnLowered = ltree.LoweredFunction
  type FnMutant  = ltree.LMutantIs
  type FnPanic   = ltree.LPanic
  type FnCall    = ltree.LFailingCall
  type FnClaim   = ltree.LClaim
  type FnWrite   = ltree.LWrite
  type FnSpec    = semantics.BFormatSpec
  type FnAssign  = semantics.BAssign
  type FnPart    = semantics.BInterpolationPart
  
  Ctx fnCompile(string source) {
~     var bytes = List<u8>()
~     for b in source.bytes {
~         bytes.append(b)
      }
~     var sources = List<lexer.SourceText>()
~     sources.append(lexer.SourceText.create("fn.byn", from: move bytes))
~     let options = FnOptions { includeTests: false }
~     return semantics.create(from: move sources, at: Path.fullPath("tmp"), by: in options)
  }
  
  string fnErrors(in Ctx c) {
~     var sb = StrBuf()
~     for d in c.diagnostics {
~         sb.append($"{d.code} {d.span.start} {d.message}\n") if !d.isWarning
      }
~     return sb.finish()
  }
  
  FuncId fnNamed(in Ctx c, for string name) {
~     var found: FuncId = -1
~     for i in 0..c.functions.length {
~         if c.functions[i].name == name && c.functions[i].body != -1 {
~             found = i
          }
      }
~     return found
  }
  
  List<i64> fnBody(in Ctx c, for string name) {
~     return t9Copy(c.bBlocks[c.functions[fnNamed(in c, for: name)].body & 0xFFFFFFFF].statements)
  }
  
  /// The lowered form of `name` as the lowerer leaves a body without lowered statements: its locals but not its parameters.
  FnLowered fnLowered(in Ctx c, for string name) {
~     let fn = fnNamed(in c, for: name)
~     var locals = List<i64>()
~     for l in c.functions[fn].locals {
~         locals.append(l) if !c.locals[l].isParameter
      }
~     return FnLowered { symbol: fn, locals: move locals, contextSlots: 0, panicSlots: 0, body: c.functions[fn].body }
  }
  
  /// A frame for `f` with its file's line starts, as `emitFunction` lends them.
  CFunc fnFrame(ref CCtx cc, from ref Ctx c, for in FnLowered f) {
~     var fe = functionEmitter(ref cc, from: ref c, for: in f)
~     fe.lineStarts = lineStartsOf(in c, for: c.functions[f.symbol].file)
~     return fe
  }
  
  string fnLines(List<string> lines) {
~     var sb = StrBuf()
~     for x in lines {
~         sb.append(x)
~         sb.append("\n")
      }
~     return sb.finish()
  }
  
  TextSpan fnNoSpan() => TextSpan { start: 0, length: 0 }
  
  string fnSource() {
~     return """
          struct Money {
              i64 cents = 0
  
              public void formatInto(ref StrBuf output, as FormatSpec spec) uses alloc {
                  output.append("m")
              }
  
              public i64 total() => cents
          }
  
          i64 one() => 1
  
          bool check(ref List<i64> xs) => xs.length > 0
  
          never stop() {
              panic("x")
          }
  
          void tuples(ref List<i64> rp, by i64 vp, to ref Map<i64, i64> m) {
              var t = (one(), [1])
              t.item1 = 2
              var xs: List<i64> = [1, 2]
              xs[0] = 3
              m[1] = 2
              var n = one()
              n += 1
              if check(ref xs) {
                  n = 2
              }
              let money = Money { cents: 1 }
              let held = money
              rp[0] = vp
          }
  
          void runs(List<string> args) {
              let r = Io.run("true", args.span) catch {
                  return
              }
          }
  
          void main() {
          }
          """
  }
  
  test "func t29: a never function ends in byn_unreachable, or returns a zero error when it fails, and has no #line without a file" {
      var c = fnCompile(fnSource())
      assert fnErrors(in c) == ""
      var cc = CCtx { ownershipChecks: true, packageRoot: c.packageRoot }
      let t = FnTree {}
      var f = fnLowered(in c, for: "stop")
      let plain = emitFunction(ref cc, from: ref c, through: in t, for: in f)
      assert plain == fnLines(
          [
              "BYN_NORETURN void byn_main__stop(void) {",
              "#line 16 \"fn.byn\"",
              "    (void)byn_panic_str(BYN_STR(\"x\", 1), byn_file_0, 16);",
              "    byn_unreachable(byn_file_0, 15);",
              "}",
              "#line generated",
          ],
      )
      // A local of type never is not declared.
      let tuples = fnNamed(in c, for: "tuples")
      let held = c.functions[tuples].locals[c.functions[tuples].locals.length - 1]
      c.locals[held].type = c.functions[f.symbol].returnType
      f.locals.append(held)
      c.functions[f.symbol].hasFails = true
      let failing = emitFunction(ref cc, from: ref c, through: in t, for: in f)
      assert failing == fnLines(
          [
              "byn_err byn_main__stop(void) {",
              "#line 16 \"fn.byn\"",
              "    (void)byn_panic_str(BYN_STR(\"x\", 1), byn_file_0, 16);",
              "    return (byn_err){0};",
              "}",
              "#line generated",
          ],
      )
      c.functions[f.symbol].hasFails = false
      c.functions[f.symbol].file = -1
      let fileless = emitFunction(ref cc, from: ref c, through: in t, for: in f)
      assert fileless == fnLines(
          [
              "BYN_NORETURN void byn_main__stop(void) {",
              "    (void)byn_panic_str(BYN_STR(\"x\", 1), byn_file_1, 0);",
              "    byn_unreachable(byn_file_1, 0);",
              "}",
          ],
      )
  }

  test "func t29: roots and places through tuple items, parameters and lowered nodes" {
      var c = fnCompile(fnSource())
      var cc = CCtx { ownershipChecks: true, packageRoot: c.packageRoot }
      var t = FnTree {}
      let mutant = ltree.addLMutantIs(ref t, from: FnMutant { span: fnNoSpan(), id: 7 })
      let f = fnLowered(in c, for: "tuples")
      var fe = fnFrame(ref cc, from: ref c, for: in f)
      let body = fnBody(in c, for: "tuples")
      let item = c.bAssigns[body[1] & 0xFFFFFFFF].target
      let tuple = c.bLocals[c.bTupleElements[item & 0xFFFFFFFF].receiver & 0xFFFFFFFF].local
      let rp = c.bIndexes[c.bAssigns[body[10] & 0xFFFFFFFF].target & 0xFFFFFFFF].target
      let vp = c.bAssigns[body[10] & 0xFFFFFFFF].value
      let xs = c.bIndexes[c.bAssigns[body[3] & 0xFFFFFFFF].target & 0xFFFFFFFF].target
      assert c.locals[tuple].name == "t"
      assert ownerRoot(c, for: item) == tuple
      assert root(c, for: item) == tuple
      assert ownerRoot(c, for: mutant) == -1
      assert root(c, for: mutant) == -1
      assert placeOf(ref cc, for: ref fe, from: ref c, through: t, at: item) == "l0_t.item1"
      assert placeOf(ref cc, for: ref fe, from: ref c, through: t, at: mutant) == "byn_mutant_is(7)"
      assert placeOf(ref cc, for: ref fe, from: ref c, through: t, at: rp) == "(*p_rp)"
      assert placeOf(ref cc, for: ref fe, from: ref c, through: t, at: vp) == "p_vp"
      assert checkedStore(ref cc, for: ref fe, from: ref c, through: t, at: mutant, to: vp) is null
      // A place passed by pointer is claimed after the statement unless the statement moves it.
      claimAfterCall(ref cc, for: ref fe, from: ref c, through: t, at: xs, to: fnNoSpan())
      assert fe.afterCalls.length == 1
      assert fe.afterCalls[0] == "bynh_chk_claim_list_i64(&l1_xs, byn_file_0, 1);"
      fe.moved[c.bLocals[xs & 0xFFFFFFFF].local] = true
      claimAfterCall(ref cc, for: ref fe, from: ref c, through: t, at: xs, to: fnNoSpan())
      assert fe.afterCalls.length == 1
  }
  
  test "func t29: assignments, an if with claims after its condition, and blocks" {
      var c = fnCompile(fnSource())
      var cc = CCtx { ownershipChecks: true, packageRoot: c.packageRoot }
      var t = FnTree {}
      let f = fnLowered(in c, for: "tuples")
      var fe = fnFrame(ref cc, from: ref c, for: in f)
      let body = fnBody(in c, for: "tuples")
      var sb = StrBuf()
      for k in 0..body.length {
          let s = body[k]
          if semantics.boundKind(s) is .assign or .ifStmt {
              sb.append($"-- {k}\n")
              statement(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: s)
          }
      }
      sb.append("-- then\n")
      statement(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: c.bIfs[body[7] & 0xFFFFFFFF].then)
      sb.append("-- block of a statement\n")
      block(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: body[1])
      let p = ltree.addLPanic(ref t, from: FnPanic { span: fnNoSpan(), message: "boom" })
      sb.append("-- block of a lowered statement\n")
      block(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: p)
      assert sb.finish() == fnLines(
          [
              "-- 1",
              "#line 21 \"fn.byn\"",
              "    l0_t.item1 = INT64_C(2);",
              "-- 3",
              "#line 23 \"fn.byn\"",

  test "func t29: Io.run's failing call, claims with nothing to claim, and a write's empty part" {
      var c = fnCompile(fnSource())
      var cc = CCtx { ownershipChecks: true, packageRoot: c.packageRoot }
      var t = FnTree {}
      let f = fnLowered(in c, for: "runs")
      var fe = fnFrame(ref cc, from: ref c, for: in f)
      var run: BExpr = -1
      for i in 0..c.bCalls.length {
          if c.functions[c.bCalls[i].function].builtin == .ioRun {
              run = semantics.makeBound(.call, at: i)
          }
      }
      let r = f.locals[0]
      let fc = ltree.addLFailingCall(ref t, from: FnCall { span: fnNoSpan(), error: r, call: run })
      let tuples = fnNamed(in c, for: "tuples")
      let held = c.functions[tuples].locals[c.functions[tuples].locals.length - 1]
      let vp = c.functions[tuples].parameters[1]
      assert c.locals[held].name == "held"
      assert c.locals[vp].name == "vp"
      let notOwned = ltree.addLClaim(ref t, from: FnClaim { span: fnNoSpan(), local: vp })
      let noStorage = ltree.addLClaim(ref t, from: FnClaim { span: fnNoSpan(), local: held })
      var parts = List<FnPart>()
      parts.append(FnPart { text: "", value: -1 })
      parts.append(FnPart { text: "a" })
      let w = ltree.addLWrite(ref t, from: FnWrite { span: fnNoSpan(), sink: .panic, parts: move parts })
      var sb = StrBuf()
      statement(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: fc)
      sb.append("-- claims\n")
      statement(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: notOwned)
      statement(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: noStorage)
      sb.append("-- write\n")
      statement(ref cc, for: ref fe, from: ref c, through: t, to: ref sb, at: w)
      assert sb.finish() == fnLines(
          [
              "#line 1 \"fn.byn\"",
              "    {",
              "        int64_t detail = 0;",
              "        byn_process proc = {0};",
              "        l0_r = (byn_err){ .code = bynh_io_run(BYN_STR(\"true\", 4), bynh_span_list_str(p_args), &proc, &detail) };",
              "        if (l0_r.code == UINT32_C(10)) l0_r.p.e10.signal = (int32_t)detail;",
              "        if (l0_r.code == UINT32_C(10)) { l0_r.p.e10.stdout_ = proc.out; l0_r.p.e10.stderr_ = proc.err; }",
              "        if (l0_r.code == UINT32_C(11)) l0_r.p.e11.after = (int64_t)detail;",
              "        if (l0_r.code == UINT32_C(11)) { l0_r.p.e11.stdout_ = proc.out; l0_r.p.e11.stderr_ = proc.err; }",
              "    }",
              "-- claims",
              "-- write",
              "#line 1 \"fn.byn\"",
              "    {",
              "        uint8_t wb[1024];",
              "        byn_writer wv = {wb, 1024, 0, BYN_WRITE_TRUNCATE, byn_file_0, 1};",
              "        byn_writer* w = &wv;",
              "        byn_w_str(w, BYN_STR(\"a\", 1));",
