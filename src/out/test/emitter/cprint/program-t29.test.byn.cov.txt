  import semantics
  import emitter.ltree
  import lexer
  
  // T29: program.byn's sections, `main` and names on a bound program; this module cannot import the lowerer, so each test
  // builds the `LoweredProgram` it emits.
  
  type PgOptions  = semantics.CompilationOptions
  type PgProgram  = ltree.LoweredProgram
  type PgLowered  = ltree.LoweredFunction
  type PgFileText = lexer.SourceText
  type PgTree     = ltree.LTree
  
  Ctx pgCompile(string source, for string path, by bool tests) {
~     var bytes = List<u8>()
~     for b in source.bytes {
~         bytes.append(b)
      }
~     var sources = List<PgFileText>()
~     sources.append(PgFileText.create(path, from: move bytes))
~     let options = PgOptions { includeTests: tests }
~     return semantics.create(from: move sources, at: Path.fullPath("tmp"), by: in options)
  }
  
  string pgErrors(in Ctx c) {
~     var sb = StrBuf()
~     for d in c.diagnostics {
~         sb.append($"{d.code} {d.span.start} {d.message}\n") if !d.isWarning
      }
~     return sb.finish()
  }
  
  FuncId pgNamed(in Ctx c, for string name, as semantics.FunctionKind kind) {
~     var found: FuncId = -1
~     for i in 0..c.functions.length {
~         if (name == "" || c.functions[i].name == name) && c.functions[i].kind == kind {
~             found = i
          }
      }
~     return found
  }
  
  test "program t29: no error variants, a coverage build with no probes" {
      var c = pgCompile("void main() {}\n", for: "pg.byn", by: false)
      assert pgErrors(in c) == ""
      c.errorVariants = []
      var cc = CCtx {}
      let t = PgTree {}
      let program = PgProgram { functions: [], main: -1, tests: [], testFunctions: [], covering: true }
      let text = t6Run(ref cc, from: ref c, through: in t, for: in program)
      assert text.contains("static const byn_str byn_errors[] = {\n    BYN_STR(\"\", 0),\n};\n")
      assert text.contains("static const uint8_t byn_exit_codes[] = {1};\n")
      assert text.contains("static uint64_t byn_cov[1];\n")
  }
  
  test "program t29: main under tests, coverage and symbols, failing with a code and arguments" {
      var c = pgCompile(
          """
          union error E {
              bad
          }
  
          i32 main(ReadOnlySpan<string> args) fails E {
              return 0
          }
  
          test "t" {}
  
          """,
          for: "pg.byn",
          by: true,
      )
      assert pgErrors(in c) == ""
      let main = pgNamed(in c, for: "main", as: .function)
      let probe = pgNamed(in c, for: "", as: .testKeyword)
      var cc = CCtx { symbols: true }
      cc.testIndex[probe] = 0
      var tests = List<PgLowered>()
      tests.append(PgLowered { symbol: probe, locals: [], contextSlots: 0, panicSlots: 0, body: c.functions[probe].body })
      let program = PgProgram { functions: [], main: main, tests: [0], testFunctions: move tests, covering: true }
      var lines = List<string>()
      lines.append("int main(int argc, char** argv) {")
      lines.append("    byn_err err = {0};")
      lines.append("    byn_cov_start(argc == 3 && strcmp(argv[1], \"--test\") == 0 ? argv[2] : NULL);")
      lines.append("    byn_hash_seed_init(argc == 3 && strcmp(argv[1], \"--test\") == 0);")
      lines.append("    byn_sym_init(byn_symtab, (uint32_t)(sizeof byn_symtab / sizeof byn_symtab[0]));")
      lines.append("    if (argc == 3 && strcmp(argv[1], \"--test\") == 0) {")
      lines.append("        switch (atoi(argv[2])) {")
      lines.append("        case 0: byn_test_0(); break;")
      lines.append("        default: return 2;")
      lines.append("        }")
      lines.append("        if (err.code != 0) { byn_report_error(byn_errors[err.code - 1]); return 1; }")
      lines.append("        byn_stdout_flush();")
      lines.append("        return 0;")
      lines.append("    }")
      lines.append("    byn_str* byn_argv = byn_args(&argc, argv);")
      lines.append("    int32_t code = 0;")
      lines.append("    err = byn_main__main((bynt_span_str){byn_argv, argc > 1 ? argc - 1 : 0}, &code);")
      lines.append(
          "    if (err.code != 0) { byn_report_error(byn_errors[err.code - 1]); return byn_exit_codes[err.code - 1]; }",
      )
      lines.append("    byn_stdout_flush();")
      lines.append("    return code;")
      lines.append("}")
      var expected = StrBuf()
      for x in lines {
          expected.append(x)
          expected.append("\n")
      }
      assert t6EmitMain(ref cc, from: ref c, through: in program) == expected.finish()
  }
  
  test "program t29: a repeated symbol text, and marks that are not a symbol index" {
      var cc = CCtx { symbols: true }
      cc.symbolTexts = ["b", "a", "b"]
      var sb = StrBuf()
      let indexes = t6SymbolTable(ref cc, to: ref sb)
      assert sb.finish() == "static const byn_str byn_symtab[] = {\n    BYN_STR(\"\", 0),\n    BYN_STR(\"a\", 1),\n    BYN_STR(\"b\", 1),\n};\n\n"
      assert $"{indexes[0]} {indexes[1]} {indexes[2]}" == "2 1 2"
      let mark = $"{rune(symbolMark)}"
      let text = $"x{mark}1{mark}y{mark}z{mark}2:{mark}{mark}2{mark}45{mark}6"
      assert t6SymbolIndexes(text, by: in indexes) == $"x1y{mark}z{mark}2:{mark}245{mark}6"
  }
  
  test "program t29: a builtin's name, and a rooted file outside the package named by its file name" {
      var c = pgCompile("void main() {}\nvoid f() {}\n", for: Path.fullPath("/elsewhere/dir/pg.byn"), by: false)
      var cc = CCtx { packageRoot: Path.fullPath("/root/package") }
      let read = pgNamed(in c, for: "readFile", as: .builtin)
      assert functionName(ref cc, from: ref c, for: read) == "byn_builtin__readFile"
      let f = pgNamed(in c, for: "f", as: .function)
      assert filePath(ref cc, from: ref c, for: f) == "pg.byn"
      assert t10CoverageCounters(0).contains("static uint64_t byn_cov[1];\n")
  }
