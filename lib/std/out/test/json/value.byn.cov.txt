  // std.json tree (spec 016 T4, 015 S7): `JsonValue` built on `JsonReader`, and `Json.encode`.
  
  /// A JSON value. A number keeps its text, so `Json.encode` writes it back as read; an object keeps its members
  /// in text order, duplicate keys included.
  public union JsonValue {
      nothing
      flag(bool value)
      number(f64 value, string text)
      text(string value)
      list(List<JsonValue> items)
      object(List<JsonMember> members)
  }
  
  /// One `key: value` of an object.
  public struct JsonMember {
      public string key
      public JsonValue value
  }
  
  /// Parsing JSON text into a tree and writing a tree as text.
  public struct Json {
      /// The value `text` holds, read with `JsonReader`; outside data, so `Untrusted`.
      public static Untrusted<JsonValue> parse(string text) fails JsonError uses alloc {
+         var reader = JsonReader.over(text)
+         let first = (try reader.next()) ?? panic("std.json: a reader gives a token before its end")
+         let value = try treeOf(ref reader, from: first)
+         let _ = try reader.next()
+         return Untrusted(move value)
      }
  
      /// `v` as text, the layout of Python's `json.dumps(v, indent=indent)` with `ensure_ascii`: compact with `, ` and
      /// `: ` when `indent` is 0 or less, else one item per line, `indent` spaces per level. A number is its text.
      public static string encode(JsonValue v, i32 indent = 0) uses alloc {
+         var sb = StrBuf()
+         encodeInto(ref sb, from: v, by: i64(indent), at: 0)
+         return $"{sb.finish()}"
      }
  }
  
  /// The value that starts with `first`; the reader's depth limit bounds the recursion.
  JsonValue treeOf(ref JsonReader reader, from JsonToken first) fails JsonError uses alloc {
+     switch first {
          case .startList:
+             var items = List<JsonValue>()
+             while true {
+                 let token = (try reader.next()) ?? panic("std.json: a reader gives a token before its end")
+                 if token is .endList { return .list(items: move items) }
+                 items.append(try treeOf(ref reader, from: token))
              }
          case .startObject:
+             var members = List<JsonMember>()
+             while true {
+                 let token = (try reader.next()) ?? panic("std.json: a reader gives a token before its end")
+                 switch token {
+                     case .endObject: return .object(members: move members)
                      case .key(let name):
+                         let next = (try reader.next()) ?? panic("std.json: a reader gives a token before its end")
+                         members.append(JsonMember { key: name, value: try treeOf(ref reader, from: next) })
-                     default: panic("std.json: a reader gives a key or `}` in an object")
                  }
              }
+         case .text(let value): return .text(value: value)
+         case .number(let value, let text): return .number(value: value, text: text)
+         case .flag(let value): return .flag(value: value)
+         case .nothing: return .nothing
-         default: panic("std.json: a reader starts a value with a value token")
      }
  }
  
  void encodeInto(ref StrBuf sb, from JsonValue v, by i64 indent, at i64 level) uses alloc {
+     switch v {
+         case .nothing: sb.append("null")
+         case .flag(let value): sb.append(value ? "true" : "false")
+         case .number(_, let text): sb.append(text)
+         case .text(let value): quoteInto(ref sb, from: value)
          case .list(let items):
+             if items.length == 0 {
+                 sb.append("[]")
+                 return
              }
+             sb.append("[")
+             for i in 0..items.length {
+                 separate(ref sb, by: indent, at: level + 1, first: i == 0)
+                 encodeInto(ref sb, from: items[i], by: indent, at: level + 1)
              }
+             newline(ref sb, by: indent, at: level)
+             sb.append("]")
          case .object(let members):
+             if members.length == 0 {
+                 sb.append("{}")
+                 return
              }
+             sb.append("{")
+             for i in 0..members.length {
+                 separate(ref sb, by: indent, at: level + 1, first: i == 0)
+                 quoteInto(ref sb, from: members[i].key)
+                 sb.append(": ")
+                 encodeInto(ref sb, from: members[i].value, by: indent, at: level + 1)
              }
+             newline(ref sb, by: indent, at: level)
+             sb.append("}")
      }
  }
  
  /// Before an item: `, ` (compact) or `,` and a new line (indented); the first item has no comma.
  void separate(ref StrBuf sb, by i64 indent, at i64 level, bool first) {
+     if !first { sb.append(indent > 0 ? "," : ", ") }
+     newline(ref sb, by: indent, at: level)
  }
  
  void newline(ref StrBuf sb, by i64 indent, at i64 level) {
+     if indent <= 0 { return }
+     sb.append("\n")
+     sb.appendRepeated(' ', by: indent * level)
  }
  
  /// `s` as a JSON string with Python's `ensure_ascii` escapes: `\uXXXX` (lowercase, a surrogate pair above U+FFFF)
  /// for every rune outside `' '..'~'` that has no short escape.
  void quoteInto(ref StrBuf sb, from string s) {
+     sb.append("\"")
+     for r in s.runes {
+         let cp = r.value
+         if r == '"' {
+             sb.append("\\\"")
+         } else if r == '\\' {
+             sb.append("\\\\")
+         } else if r == '\n' {
+             sb.append("\\n")
+         } else if r == '\r' {
+             sb.append("\\r")
+         } else if r == '\t' {
+             sb.append("\\t")
+         } else if cp == 0x8 {
+             sb.append("\\b")
+         } else if cp == 0xC {
+             sb.append("\\f")
+         } else if cp >= 0x20 && cp <= 0x7E {
+             sb.append($"{r}")
+         } else if cp > 0xFFFF {
+             let v = cp - 0x10000
+             unitInto(ref sb, from: 0xD800 + v / 0x400)
+             unitInto(ref sb, from: 0xDC00 + v % 0x400)
          } else {
+             unitInto(ref sb, from: cp)
          }
      }
+     sb.append("\"")
  }
  
  void unitInto(ref StrBuf sb, from u32 unit) {
+     sb.append("\\u")
+     let digits = string(unit, as: .hex)
+     sb.appendRepeated('0', by: 4 - digits.length)
+     sb.append(digits)
  }
